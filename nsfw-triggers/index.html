Gag trigger​

name: "equipped_zbf_gag_ball_trigger"
description: "Trigger on equip event "

eventCriteria:
  eventType: "equip"
  schemaConditions:
    - fieldPath: "item"
      operator: "equals"
      value: "zbf Gag Ball"
    - fieldPath: "action"
      operator: "equals"
      value: "equipped"

response:
  type: "direct_narration"
  content: "{{ event_json.actor }} has a {{ event_json.item }} put on the mouth, making {{ event_json.actor }} unable to speak"

audience: "nearby_npcs"

enabled: true
interrupt: false
probability: 1
cooldownSeconds: 30
priority: 1
